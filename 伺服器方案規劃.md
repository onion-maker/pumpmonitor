// server/src/tide.ts
export function detectTideDirection(levelOutArray: number[]): 'rising' | 'falling' | 'slack' {
  const last5 = levelOutArray.slice(-5);
  if (last5.length < 3) return 'slack';

  let inc = 0, dec = 0;
  for (let i = 1; i < last5.length; i++) {
    if (last5[i] > last5[i-1]) inc++;
    else if (last5[i] < last5[i-1]) dec++;
  }

  if (dec > inc) return 'falling';   // 退潮
  if (inc > dec) return 'rising';    // 漲潮
  return 'slack';
}

// 使用 3-record trend 作為 speedy 判斷
export function detectTideTrend3(records: TideRecord[]): 'rising' | 'falling' | 'slack' {
  const recent = records.slice(-3);
  if (recent.length < 3) return 'slack';

  const headAvg = (recent[0].level_out + recent[1].level_out) / 2;
  const tailAvg = (recent[1].level_out + recent[2].level_out) / 2;

  if (tailAvg < headAvg) return 'falling';
  if (tailAvg > headAvg) return 'rising';
  return 'slack';
}

7.2 Gate Alarm Logic

對應到原前端條件：

// server/src/alarm.ts
export function checkTideGate(records: TideRecord[], prevDoor: DoorState[]): string | null {
  const trend = detectTideTrend3(records);
  const { level_out, level_in } = records[records.length - 1];

  // 開閘：退潮 AND level_out < level_in AND 所有閘門關閉
  if (trend === 'falling' && level_out < level_in && prevDoor.every(d => d !== '2')) {
    return 'tide_open_gate';
  }

  // 關閉閘：漲潮 AND 任一閘門開啟
  if (trend === 'rising' && prevDoor.some(d => d === '2')) {
    return 'tide_close_gate';
  }

  return null;
}

結論

移植這套系統到 GCP e2-micro + FCM 可以100% 保留現有演算法，只需要：

1. 把所有 fetchPumpInvestigationAPI 移到 Node.js（38 個 API 呼叫）
2. 把 tide detection + alarm logic 移到 Node.js（單一 source of truth）
3. 改造 App 成「server-only data + FCM notification」
4. 移除 Android 後台服務

打個比方：你現在有兩個「警報閱覽員」看電視，要改成「一個在辦公室看電視、手機只能收通知」——只要把辦公室的那位改成 Node.js server，然後手機的電視盒子只顯示 + 接收推播，就完成了。
